PNA(
    node_features_dim: Union[int, Tuple[int, int]],
    hidden_features_dim: int,
    num_layers: int,
    out_features_dim: Optional[int] = None,
    dropout: float = 0.0,
    act: Activation = "relu",
    norm: Normalization = None,
    jk: Optional[Literal["cat", "max", "lstm"]] = None,
    **kwargs: Any,
)

Bases: BasicGNN

Principal Neighbourhood Aggregation network from "Principal Neighbourhood Aggregation for Graph Nets", a stack of PNAConv layers.

Parameters:

Name Type Description Default
node_features_dim Union[int, Tuple[int, int]]

Size of the input node features.

 required
hidden_features_dim int

Size of the hidden node embeddings.

 required
num_layers int

Number of message passing layers.

 required
out_features_dim Optional[int]

Size of the output. Default: None

 None
dropout float

Dropout probability. Default: 0.0

 0.0
act Activation

Activation between layers. Default: "relu"

 'relu'
norm Normalization

Normalization between layers. Default: None

 None
jk Optional[Literal['cat', 'max', 'lstm']]

Jumping Knowledge mode. Default: None

 None
**kwargs Any

Arguments of PNAConv; aggregators, scalers and deg are required (e.g. also towers or edge_features_dim).

 {}

Example:

from mlx_graphs.nn import PNA, PNAConv

deg = PNAConv.get_degree_histogram(train_dataset)
model = PNA(
    dataset.num_node_features,
    64,
    num_layers=3,
    out_features_dim=dataset.num_graph_classes,
    aggregators=["mean", "min", "max", "std"],
    scalers=["identity", "amplification", "attenuation"],
    deg=deg,
)
h = model(graph.edge_index, graph.node_features)

def __init__(
    self,
    node_features_dim: Union[int, Tuple[int, int]],
    hidden_features_dim: int,
    num_layers: int,
    out_features_dim: Optional[int] = None,
    dropout: float = 0.0,
    act: Activation = "relu",
    norm: Normalization = None,
    jk: Optional[Literal["cat", "max", "lstm"]] = None,
    **kwargs: Any,
):
    super().__init__()
    if num_layers < 1:
        raise ValueError("`num_layers` must be at least 1")

    self.node_features_dim = node_features_dim
    self.hidden_features_dim = hidden_features_dim
    self.num_layers = num_layers
    self.out_features_dim = (
        out_features_dim if out_features_dim is not None else hidden_features_dim
    )
    self.act = _resolve_activation(act)
    # Normalization name, for convolutions with inner MLPs (`init_conv`).
    self._norm_name = norm
    self.dropout = nn.Dropout(dropout) if dropout > 0.0 else None
    self._dropout_p = dropout

    # Without Jumping Knowledge the last convolution maps directly to the
    # output size; with it, every convolution outputs hidden embeddings and
    # a final linear layer produces the output.
    conv_out_last = hidden_features_dim
    if out_features_dim is not None and jk is None:
        conv_out_last = out_features_dim
    self._plain_last = out_features_dim is not None and jk is None

    # As in PyG, a (src, dst) input size gives every layer separate
    # source and destination sizes.
    hidden_in: Union[int, Tuple[int, int]] = hidden_features_dim
    if isinstance(node_features_dim, (tuple, list)):
        node_features_dim = tuple(node_features_dim)
        hidden_in = (hidden_features_dim, hidden_features_dim)
    in_dims = [node_features_dim] + [hidden_in] * (num_layers - 1)
    out_dims = [hidden_features_dim] * (num_layers - 1) + [conv_out_last]
    self.convs = [
        self.init_conv(in_dim, out_dim, is_last=(i == num_layers - 1), **kwargs)
        for i, (in_dim, out_dim) in enumerate(zip(in_dims, out_dims))
    ]

    num_post = num_layers - 1 if self._plain_last else num_layers
    if norm is not None:
        self.norms = [
            _make_norm(norm, hidden_features_dim) for _ in range(num_post)
        ]

    if jk is not None:
        self.jk = JumpingKnowledge(jk, hidden_features_dim, num_layers)
        jk_dim = hidden_features_dim * (num_layers if jk == "cat" else 1)
        if out_features_dim is not None or jk == "cat":
            self.lin = Linear(jk_dim, self.out_features_dim)
